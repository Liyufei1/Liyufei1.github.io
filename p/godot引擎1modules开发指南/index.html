Modules开发指南

本文档是针对Godot 4.4引擎的 Modules开发指南，为需要进行底层开发的开发者提供实用的参考。Godot Modules开发允许我们直接在引擎源码层面扩展功能，相比GDExtension具有更强大的能力和更深度的集成。本文涵盖了日志打印、容器类型、GUI组件、线程管理、配置系统等核心模块的常用头文件和最小使用示例，帮助你快速上手 Godot 引擎的 C++ 开发。

自定义项目配置

头文件：#include “core/config/project_settings.h”

ProjectSettings是 Godot引擎的核心单例类，负责管理项目的配置和设置。它继承自Object，使用 GDCLASS 宏注册到 Godot 的类型系统中
核心职责：

  • 加载和保存项目配置文件（project.godot）
    • 定义全局设置（带默认值）:GLOBAL_DEF
    • 定义为基础设置（在编辑器中高亮）:GLOBAL_DEF_BASIC
    • 定义全局设置（修改后需要重启）:GLOBAL_DEF_RST
    • 定义为内部设置（不在编辑器显示）:GLOBAL_DEF_INTERNAL
    • 获取配置值: get_setting
    • 设置配置值: set_setting
    • 检查配置值是否存在: has_setting
  • 路径转换（res://、user:// 等路径本地化）
    • 将绝对路径转换为 res:// 路径: localize_path
    • 将 res:// 路径转换为绝对路径: globalize_path
    • 获取项目资源路径（res:// 的实际位置）: get_resource_path
    • 获取项目数据目录路径（res://.godot): get_project_data_path
    • 获取导入文件路径（res://.godot/imported）: get_imported_files_path
  • Autoload 单例管理
    • 添加自动加载项:add_autoload
    • 移除自动加载项:remove_autoload
    • 检查是否存在:has_autoload
    • 获取自动加载列表:get_autoload_list

#include "core/variant/array.h"
#include "core/variant/variant.h"

void demo_array() {
    // --- 创建 (Create) ---
    Array arr;

    // 也可以创建并指定为特定类型的数组（例如只存 int）
    // Array typed_arr;
    // typed_arr.set_typed(Variant::INT, StringName(), Variant());

    // --- 增 (Add) ---
    arr.push_back(10);          // 自动封装为 Variant(10)
    arr.append("Hello");        // 自动封装为 Variant("Hello")
    arr.push_front(3.14);       // 插入到头部
    
    // 此时数组内容可能为: [3.14, 10, "Hello"]

    // --- 查 (Read) ---
    if (!arr.is_empty()) {
        // operator[] 返回的是 Variant &
        Variant v = arr[0]; 
        
        // 需要根据类型转换
        if (v.get_type() == Variant::FLOAT) {
            double d = v; // 自动转型操作符
            print_line(rtos(d));
        }
    }

    // 遍历
    for (int i = 0; i < arr.size(); i++) {
        Variant item = arr[i];
        // print_line(item); // Variant 重载了 String 转换
    }

    // --- 改 (Update) ---
    arr[1] = "New Value"; // 将索引 1 的 10 改为字符串

    // --- 删 (Delete) ---
    arr.remove_at(0); // 移除第一个元素
    arr.erase("New Value"); // 移除匹配值的元素（线性查找）
    arr.clear();
}

#include "core/variant/dictionary.h"
#include "core/variant/variant.h"

void demo_dictionary() {
    // --- 创建 (Create) ---
    Dictionary dict;

    // --- 增 (Add) / 改 (Update) ---
    dict["name"] = "Godot";      // 键是 String, 值是 String
    dict[100] = "One Hundred";   // 键是 int, 值是 String
    
    // 也可以作为左值修改
    dict["name"] = "Godot Engine";

    // --- 查 (Read) ---
    // 检查键是否存在
    if (dict.has("name")) {
        Variant v = dict["name"];
        String s = v; 
        print_line(s);
    }

    // 安全获取（带默认值）
    // 如果 "score" 不存在，返回 0
    int score = dict.get("score", 0);

    // 遍历
    // 注意：遍历字典通常遍历的是 Key
    List<Variant> keys;
    dict.get_key_list(&keys); // 获取所有 key 到一个 List 中
    
    for (const Variant &key : keys) {
        Variant value = dict[key];
        print_line(String(key) + " : " + String(value));
    }

    // --- 删 (Delete) ---
    dict.erase(100); // 删除键为 100 的条目
    bool removed = dict.erase("name"); // 返回是否删除成功
}

#include "core/templates/vmap.h"

void demo_vmap() {
    VMap<int, String> player_names;

    // --- 增 (Add) ---
    // 注意：insert 会检查键是否存在，存在则覆盖，不存在则保持排序插入
    player_names.insert(1, "Alice");
    player_names.insert(2, "Bob");
    player_names.insert(10, "Boss");

    // --- 查 (Read) ---
    // 检查是否存在
    if (player_names.has(2)) {
        // 获取值，如果 key 不存在会 crash，所以最好先检查 has
        String name = player_names.get(2); 
        
        // 或者使用 getptr，如果不存在返回 nullptr
        const String *name_ptr = player_names.getptr(2);
    }
    
    // 查找索引（用于遍历内部 vector）
    int idx = player_names.find(10); // 返回内部 vector 的 index
    if (idx != -1) {
        // 通过 vector 索引访问
        String val = player_names.get_value_at_index(idx);
    }

    // --- 改 (Update) ---
    player_names.insert(1, "AliceUpdated"); // insert 兼具更新功能
    
    // 如果你有索引，也可以通过索引改
    // player_names.get_value_at_index(0) = "NewName";

    // --- 删 (Delete) ---
    player_names.erase(10); // 移除 key 为 10 的元素
}

#include "core/templates/list.h"
#include "core/string/ustring.h"

void demo_list() {
    // --- 1. 创建 (Create) ---
    List<String> chat_log;

    // --- 2. 增 (Add) ---
    // 尾部添加 O(1)
    chat_log.push_back("Player joined"); 
    chat_log.push_back("Player said: Hello");
    
    // 头部添加 O(1) - 适合做队列
    chat_log.push_front("System: Welcome"); 

    // 在特定位置插入
    // List 的插入通常依赖于 Element 指针
    List<String>::Element *E = chat_log.front(); // 获取第一个节点 ("System: Welcome")
    if (E) {
        // 在第一个节点后面插入
        chat_log.insert_after(E, "System: Message of the day");
    }

    // 此时顺序: 
    // [0] System: Welcome
    // [1] System: Message of the day
    // [2] Player joined
    // [3] Player said: Hello

    // --- 3. 查 (Read) ---
    
    // A. 访问头尾 O(1)
    if (!chat_log.is_empty()) {
        String first = chat_log.front()->get();
        String last = chat_log.back()->get();
    }

    // B. 遍历 (推荐方式 1: 范围 for 循环 - Godot 4.x)
    for (const String &msg : chat_log) {
        print_line("Log: " + msg);
    }

    // C. 遍历 (传统方式/手动方式: 使用 Element 指针)
    // 这种方式在你需要操作节点(如删除当前节点)时很有用
    for (List<String>::Element *E = chat_log.front(); E; E = E->next()) {
        // E 是节点指针，E->get() 获取值
        if (E->get() == "Player joined") {
            print_line("Found join event!");
        }
    }

    // D. 随机访问 (警告：极慢！)
    // List 重载了 []，但它是 O(N) 的，不建议在循环中使用
    // String val = chat_log[2]; // 性能差，慎用

    // --- 4. 改 (Update) ---
    // 通过节点指针直接修改
    List<String>::Element *msg_node = chat_log.back();
    if (msg_node) {
        msg_node->get() = "Player said: Bye"; // 修改值
    }

    // --- 5. 删 (Delete) ---
    
    // A. 移除头尾 O(1)
    chat_log.pop_front(); // 移除第一个
    chat_log.pop_back();  // 移除最后一个

    // B. 移除特定值 O(N) - 需要查找
    chat_log.erase("Player joined");

    // C. 通过迭代器移除 O(1) - 最高效的中间删除方式
    List<String>::Element *node_to_delete = chat_log.front();
    // 假设我们找到了要删的节点指针
    if (node_to_delete) {
        chat_log.erase(node_to_delete); // 直接把这个节点摘除
    }

    // 清空
    chat_log.clear();
}

#include "core/templates/hash_map.h"

void demo_hashmap() {
    HashMap<String, int> inventory;

    // --- 增 (Add) / 改 (Update) ---
    inventory.insert("Apple", 5);
    inventory["Banana"] = 10; // 类似 std::map，如果不存在则创建，存在则更新
    inventory.insert("Apple", 8); // 更新 Apple 的值为 8

    // --- 查 (Read) ---
    if (inventory.has("Banana")) {
        int count = inventory["Banana"]; // 获取值
    }

    // 使用 find 获取迭代器/指针，避免两次哈希计算
    HashMap<String, int>::Iterator E = inventory.find("Apple");
    if (E) {
        int val = E->value;
        // E->key 获取键
    }
    
    // 遍历
    for (const KeyValue<String, int> &E : inventory) {
        print_line(E.key + ": " + itos(E.value));
    }

    // --- 删 (Delete) ---
    inventory.erase("Banana"); // 如果存在则删除
    bool removed = inventory.remove("Apple"); // remove 也是删除
}

#include "core/os/thread.h"
#include "core/os/mutex.h"
#include "core/os/semaphore.h"

class ThreadExample : public Node {
    GDCLASS(ThreadExample, Node);

private:
    Thread *worker_thread;
    Mutex *mutex;
    Semaphore *semaphore;
    bool thread_active;
    Vector<int> shared_data;

public:
    ThreadExample() {
        worker_thread = memnew(Thread);
        mutex = memnew(Mutex);
        semaphore = memnew(Semaphore);
        thread_active = true;
    }

    ~ThreadExample() {
        thread_active = false;
        semaphore->post(); // 唤醒线程使其退出
        if (worker_thread->is_started()) {
            worker_thread->wait_to_finish();
        }
        memdelete(worker_thread);
        memdelete(mutex);
        memdelete(semaphore);
    }

    // 线程工作函数
    static void _thread_function(void *p_userdata) {
        ThreadExample *instance = static_cast<ThreadExample *>(p_userdata);

        while (instance->thread_active) {
            // 等待信号（等待任务）
            instance->semaphore->wait();

            if (!instance->thread_active) {
                break;
            }

            // 执行耗时任务（例如数据处理）
            Vector<int> local_data;
            
            {
                // 加锁访问共享数据
                MutexLock lock(instance->mutex);
                local_data = instance->shared_data;
            }

            // 处理数据（不涉及 GUI 操作）
            for (int i = 0; i < local_data.size(); i++) {
                local_data[i] *= 2; // 示例：数据翻倍
                OS::get_singleton()->delay_usec(1000); // 模拟耗时操作
            }

            // 处理完成后，通知主线程更新
            instance->call_deferred("_on_thread_complete", local_data);
        }
    }

    // 启动线程
    void start_thread() {
        ERR_FAIL_COND(worker_thread->is_started());
        worker_thread->start(_thread_function, this);
    }

    // 添加任务（从主线程调用）
    void add_task(const Vector<int> &data) {
        MutexLock lock(mutex);
        shared_data = data;
        semaphore->post(); // 唤醒工作线程
    }

    // 线程完成回调（在主线程执行）
    void _on_thread_complete(const Vector<int> &result) {
        // 可以安全地操作 GUI
        print_line("Thread completed with " + itos(result.size()) + " items");
        
        // 更新 UI 等
        for (int i = 0; i < result.size(); i++) {
            print_line("Result[" + itos(i) + "] = " + itos(result[i]));
        }
    }

protected:
    static void _bind_methods() {
        ClassDB::bind_method(D_METHOD("start_thread"), &ThreadExample::start_thread);
        ClassDB::bind_method(D_METHOD("add_task", "data"), &ThreadExample::add_task);
        ClassDB::bind_method(D_METHOD("_on_thread_complete", "result"), &ThreadExample::_on_thread_complete);
    }
};

doc_classes

doc_classes 这一块本质上只做三件事：注册类、声明文档导出、提供 XML 文档。三者缺一不可。

1. 注册类

    GDCLASS(MyDemoNode, Node);

public:
    int add_value(int a, int b);

protected:
    static void _bind_methods() {
        ClassDB::bind_method(D_METHOD("add_value", "a", "b"), &MyDemoNode::add_value);
    }
};

#include "my_demo_node.h"

    if (p_level != MODULE_INITIALIZATION_LEVEL_SCENE) {
        return;
    }

    ClassDB::register_class<MyDemoNode>();
}

这里要区分两件事：

  • _bind_methods()：绑定方法
  • ClassDB::register_class<T>()：注册类

只绑定方法、不注册类，文档不会对应到实际可用类型。

2. 声明文档导出

在模块的 config.py 中声明文档类和文档目录：

	return "doc_classes"

要求：

  • get_doc_classes() 中的类名必须和 ClassDB 注册名一致
  • get_doc_path() 一般返回模块下的 doc_classes

缺少这一步，XML 文件不会被文档系统收集。

  • ClassDB 注册名：MyDemoNode

对工具类文档，建议优先写清楚：

  • 类职责
  • 主入口方法
  • 参数与返回值语义
  • 线程或生命周期限制
  • 副作用

检查清单

写完后至少检查 5 项：

  • 类是否在 register_types.cpp 中注册
  • 类名是否出现在 get_doc_classes() 中
  • get_doc_path() 是否指向 doc_classes
  • XML 文件名是否和类名一致
  • XML 中的方法名、参数名是否和 _bind_methods() 一致

小结

doc_classes 不是单独写一个 XML 文件就结束了，而是模块注册链路的一部分：

  1. ClassDB 能找到类
  2. config.py 能导出文档
  3. doc_classes 能提供类说明

对开发者工具类来说，重点不是描述得多详细，而是接口名、参数名、限制条件必须准确。
